# Налаштування гри My Base.
# Читаються з minetest.conf під час старту сервера.

#    У креативному режимі гравці копають майже миттєво й мають
#    необмежені ресурси.
creative_mode (Creative mode) bool false

#    Охолодження лави: коли лава торкається води — стає каменем.
#    Трохи навантажує ABM, але приємно для гри.
enable_lavacooling (Lavacooling) bool true

#    Якщо увімкнено, гравцям при першому вході видається
#    базовий набір: стальні інструменти, факели, кругляк.
give_initial_stuff (Give initial items) bool false

#    Якщо увімкнено — використовується вбудований у рушій пошук спавну,
#    який не перевіряє біом. Швидше, ніж стандартний.
engine_spawn (Use engine spawn search) bool false

#    Якщо увімкнено — логуються дії не-гравців (мобів, механізмів тощо).
#    На сервері краще false, щоб не забивати debug.txt.
log_non_player_actions (Log non-player action) bool false
